async function cargarDatos() { const competidor = document.getElementById("competidorSelect").value; const cacheKey = obtenerClaveCache(); // 1. SIEMPRE intentamos cargar primero lo que tenemos guardado localmente en el teléfono // Esto garantiza que si estás sin internet, la app abre al instante sin errores. cargarDesdeCache(cacheKey); // Si tenemos productos locales, mostramos la interfaz de una vez if (todosLosProductos.length > 0) { aplicarPendientesLocalesAData(); generarBotonesCategorias(); actualizarContadores(); aplicarFiltros(); actualizarBadgePendientes(); } else { // Si es la primera vez que se abre la app y no hay caché local, mostramos estado de carga document.getElementById("productsContainer").innerHTML = `

Cargando productos...

`; } // 2. Si hay internet, consultamos en segundo plano para refrescar la lista por si hubo cambios en la nube if (navigator.onLine) { try { const response = await fetch(`${APPS_SCRIPT_URL}?competidor=${encodeURIComponent(competidor)}&_t=${Date.now()}`); if (response.ok) { const datosNuevos = await response.json(); if (Array.isArray(datosNuevos) && datosNuevos.length > 0) { todosLosProductos = datosNuevos; localStorage.setItem(cacheKey, JSON.stringify(todosLosProductos)); // Volvemos a aplicar pendientes por si tenías cambios locales sin subir aplicarPendientesLocalesAData(); generarBotonesCategorias(); actualizarContadores(); aplicarFiltros(); } } } catch (error) { console.warn("No se pudo actualizar desde la red, usando modo local:", error); } } } function cargarDesdeCache(cacheKey) { const localData = localStorage.getItem(cacheKey); if (localData) { try { todosLosProductos = JSON.parse(localData); } catch (e) { todosLosProductos = []; } } else { todosLosProductos = []; } }
